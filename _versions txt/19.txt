def get_screen_size():
    """Get actual screen size accounting for DPI scaling"""
    user32 = ctypes.windll.user32
    user32.SetProcessDPIAware()
    width = user32.GetSystemMetrics(0)
    height = user32.GetSystemMetrics(1)
    return width, height


def get_mouse_position():
    """Get the current mouse cursor position"""
    from ctypes import wintypes
    point = wintypes.POINT()
    ctypes.windll.user32.GetCursorPos(ctypes.byref(point))
    return point.x, point.y


def get_monitor_at_cursor():
    """
    Returns the monitor info dict for the screen where the mouse cursor is located.
    Uses screeninfo to enumerate monitors and checks which one contains the cursor.
    Returns dict with keys: x, y, width, height, name
    """
    mouse_x, mouse_y = get_mouse_position()
    monitors = get_monitors()
    
    for monitor in monitors:
        # Check if mouse is within this monitor's bounds
        if (monitor.x <= mouse_x < monitor.x + monitor.width and
            monitor.y <= mouse_y < monitor.y + monitor.height):
            return {
                'x': monitor.x,
                'y': monitor.y,
                'width': monitor.width,
                'height': monitor.height,
                'name': monitor.name
            }
    
    # Fallback to primary monitor if cursor position doesn't match any
    # (shouldn't happen but defensive coding)
    for monitor in monitors:
        if monitor.is_primary:
            return {
                'x': monitor.x,
                'y': monitor.y,
                'width': monitor.width,
                'height': monitor.height,
                'name': monitor.name
            }
    
    # Last fallback: return first monitor
    if monitors:
        monitor = monitors[0]
        return {
            'x': monitor.x,
            'y': monitor.y,
            'width': monitor.width,
            'height': monitor.height,
            'name': monitor.name
        }
    
    # Ultimate fallback if no monitors found (use get_screen_size)
    width, height = get_screen_size()
    return {'x': 0, 'y': 0, 'width': width, 'height': height, 'name': 'default'}


class SnippingTool:
    """Overlay window for selecting screen region"""
    
    def __init__(self, callback):
        self.callback = callback
        self.start_x = None
        self.start_y = None
        self.current_x = None
        self.current_y = None
        self.root = None
        self.canvas = None
        self.rect = None
        self.screenshot = None
        self.photo_image = None
        self.monitor_info = None  # Store info about the active monitor
    
    def start(self):
        """Start the snipping overlay"""
        # Detect which monitor the mouse cursor is on
        self.monitor_info = get_monitor_at_cursor()
        mon_x = self.monitor_info['x']
        mon_y = self.monitor_info['y']
        mon_width = self.monitor_info['width']
        mon_height = self.monitor_info['height']
        
        current_time = datetime.datetime.now().strftime("%H:%M:%S")
        print(f"[{current_time}] Opening snipping tool on monitor: {self.monitor_info['name']} ({mon_width}x{mon_height} at {mon_x},{mon_y})")
        
        # Take screenshot of ALL screens first (ImageGrab.grab with bbox 
        # doesn't handle negative coordinates properly for multi-monitor setups)
        full_screenshot = ImageGrab.grab(all_screens=True)
        
        # Calculate the virtual screen bounds (minimum x,y across all monitors)
        monitors = get_monitors()
        virtual_left = min(m.x for m in monitors)
        virtual_top = min(m.y for m in monitors)
        
        # Convert monitor coordinates to virtual screen image coordinates
        # The full screenshot starts at (virtual_left, virtual_top)
        crop_x1 = mon_x - virtual_left
        crop_y1 = mon_y - virtual_top
        crop_x2 = crop_x1 + mon_width
        crop_y2 = crop_y1 + mon_height
        
        # Crop to get only this monitor's portion
        self.screenshot = full_screenshot.crop((crop_x1, crop_y1, crop_x2, crop_y2))
        
        print(f"[{current_time}] Virtual screen offset: ({virtual_left},{virtual_top}), Crop region: ({crop_x1},{crop_y1}) to ({crop_x2},{crop_y2})")
        
        # Create fullscreen overlay on the specific monitor
        self.root = tk.Tk()
        self.root.withdraw()  # Hide initially
        
        # Make DPI aware
        try:
            self.root.tk.call('tk', 'scaling', 1.0)
        except:
            pass
        
        # Remove window decorations and position on the correct monitor
        self.root.overrideredirect(True)
        self.root.geometry(f"{mon_width}x{mon_height}+{mon_x}+{mon_y}")
        self.root.attributes('-topmost', True)
        self.root.config(cursor="cross")
        
        # Create canvas with screenshot as background
        self.canvas = tk.Canvas(
            self.root,
            width=mon_width,
            height=mon_height,
            highlightthickness=0
        )
        self.canvas.pack()
        
        # Create darkened version of screenshot for overlay effect
        darkened = self.screenshot.copy()
        darkened = Image.blend(darkened, Image.new('RGB', darkened.size, (0, 0, 0)), 0.4)
        
        # Resize to match monitor dimensions if different (handles DPI scaling)
        if darkened.size != (mon_width, mon_height):
            darkened = darkened.resize((mon_width, mon_height), Image.Resampling.LANCZOS)
        
        self.photo_image = ImageTk.PhotoImage(darkened)
        self.canvas.create_image(0, 0, anchor=tk.NW, image=self.photo_image)
        
        # Bind events
        self.canvas.bind('<Button-1>', self.on_mouse_down)
        self.canvas.bind('<B1-Motion>', self.on_mouse_drag)
        self.canvas.bind('<ButtonRelease-1>', self.on_mouse_up)
        self.root.bind('<Escape>', self.on_escape)
        
        # Instructions text
        self.canvas.create_text(
            mon_width // 2, 30,
            text="Click and drag to select area. Press ESC to cancel.",
            fill='white',
            font=('Arial', 16, 'bold')
        )
        
        # Show window
        self.root.deiconify()
        self.root.attributes('-topmost', True)
        self.root.lift()
        self.root.focus_force()
        
        # Apply WS_EX_LAYERED style like Windows Snipping Tool
        # This ensures the window is in a special composition layer
        try:
            hwnd = self.root.winfo_id()
            GWL_EXSTYLE = -20
            WS_EX_LAYERED = 0x00080000
            WS_EX_TOPMOST = 0x00000008
            
            # Get current extended style
            ex_style = ctypes.windll.user32.GetWindowLongA(hwnd, GWL_EXSTYLE)
            
            # Add WS_EX_LAYERED and WS_EX_TOPMOST
            new_style = ex_style | WS_EX_LAYERED | WS_EX_TOPMOST
            ctypes.windll.user32.SetWindowLongA(hwnd, GWL_EXSTYLE, new_style)
            
            # Set the window to be fully opaque (255 = opaque)
            # LWA_ALPHA = 0x00000002
            ctypes.windll.user32.SetLayeredWindowAttributes(hwnd, 0, 255, 0x00000002)
        except Exception as e:
            print(f"Could not set WS_EX_LAYERED: {e}")
        
        # Capture all input (Modal)
        try:
            self.root.grab_set_global()
        except Exception:
            self.root.grab_set()
        
        # Start the keep-on-top loop
        self.force_topmost()

        self.root.mainloop()

    def force_topmost(self):
        """Aggressively keep window on top - nuclear option for msg dialogs"""
        try:
            # Use the direct window handle
            target_hwnd = self.root.winfo_id()
            
            # Constants
            GWL_EXSTYLE = -20
            WS_EX_TOPMOST = 0x00000008
            HWND_TOPMOST = -1
            HWND_NOTOPMOST = -2
            SWP_NOMOVE = 0x0002
            SWP_NOSIZE = 0x0001
            SWP_SHOWWINDOW = 0x0040
            
            # Check if we're the foreground window
            current_fg = ctypes.windll.user32.GetForegroundWindow()
            
            # 1. Force WS_EX_TOPMOST style bit
            ex_style = ctypes.windll.user32.GetWindowLongA(target_hwnd, GWL_EXSTYLE)
            if not (ex_style & WS_EX_TOPMOST):
                ctypes.windll.user32.SetWindowLongA(target_hwnd, GWL_EXSTYLE, ex_style | WS_EX_TOPMOST)
            
            # 2. If something else is in foreground, aggressively reclaim it
            if current_fg != target_hwnd:
                # First try to demote the intruder
                try:
                    ctypes.windll.user32.SetWindowPos(
                        current_fg,
                        HWND_NOTOPMOST,
                        0, 0, 0, 0,
                        SWP_NOMOVE | SWP_NOSIZE
                    )
                except:
                    pass
            
            # 3. Multiple aggressive calls to ensure we're on top
            ctypes.windll.user32.BringWindowToTop(target_hwnd)
            ctypes.windll.user32.SetWindowPos(
                target_hwnd, 
                HWND_TOPMOST, 
                0, 0, 0, 0, 
                SWP_NOMOVE | SWP_NOSIZE | SWP_SHOWWINDOW
            )
            ctypes.windll.user32.SetForegroundWindow(target_hwnd)
            
            # Tkinter-level enforcement
            self.root.lift()
            self.root.attributes('-topmost', True)
            self.root.focus_force()
            
        except Exception:
            pass
            
        # Re-schedule VERY frequently (2ms) to combat msg dialogs
        if self.root:
            self.root.after(2, self.force_topmost)
    
    def on_mouse_down(self, event):
        """Mouse button pressed"""
        self.start_x = event.x
        self.start_y = event.y
        
        # Create rectangle
        self.rect = self.canvas.create_rectangle(
            self.start_x, self.start_y, self.start_x, self.start_y,
            outline='#00ff00', width=2
        )
    
    def on_mouse_drag(self, event):
        """Mouse is being dragged"""
        self.current_x = event.x
        self.current_y = event.y
        
        # Update rectangle
        if self.rect:
            self.canvas.coords(
                self.rect,
                self.start_x, self.start_y,
                self.current_x, self.current_y
            )
    
    def on_mouse_up(self, event):
        """Mouse button released - capture the region"""
        self.current_x = event.x
        self.current_y = event.y
        
        # Calculate the bounding box (handle any direction of drag)
        x1 = min(self.start_x, self.current_x)
        y1 = min(self.start_y, self.current_y)
        x2 = max(self.start_x, self.current_x)
        y2 = max(self.start_y, self.current_y)
        
        # Close the overlay
        self.root.destroy()
        
        # Check if selection is valid (minimum size)
        if x2 - x1 > 10 and y2 - y1 > 10:
            # Get the actual screenshot dimensions
            screenshot_width, screenshot_height = self.screenshot.size
            
            # Use monitor dimensions for scaling (since we only captured this monitor)
            mon_width = self.monitor_info['width']
            mon_height = self.monitor_info['height']
            
            # Calculate scaling factors if screenshot size differs from monitor size
            scale_x = screenshot_width / mon_width
            scale_y = screenshot_height / mon_height
            
            # Apply scaling to coordinates (coordinates are local to the monitor)
            actual_x1 = int(x1 * scale_x)
            actual_y1 = int(y1 * scale_y)
            actual_x2 = int(x2 * scale_x)
            actual_y2 = int(y2 * scale_y)
            
            # Crop the screenshot to the selected region
            cropped = self.screenshot.crop((actual_x1, actual_y1, actual_x2, actual_y2))
            self.callback(cropped)
        else:
            current_time = datetime.datetime.now().strftime("%H:%M:%S")
            print("[{}] Selection too small, cancelled.".format(current_time))
    
    def on_escape(self, event):
        """ESC pressed - cancel"""
        self.root.destroy()
        current_time = datetime.datetime.now().strftime("%H:%M:%S")
        print("[{}] Snipping cancelled.".format(current_time))


def enhance_image(image):
    # Upscale for better detail
    image = upscale_and_set_dpi(image, scale_factor=2.0, dpi=300)

    # Convert PIL Image to OpenCV format
    img_cv = cv2.cvtColor(np.array(image), cv2.COLOR_RGB2BGR)

    # Convert to grayscale - Tesseract works well with grayscale
    gray = cv2.cvtColor(img_cv, cv2.COLOR_BGR2GRAY)

    # Detect if dark background (code editors often have dark themes)
    # Use median of edge pixels to detect background color
    edge_pixels = np.concatenate([
        gray[0, :], gray[-1, :],  # top and bottom rows
        gray[:, 0], gray[:, -1]   # left and right columns
    ])
    bg_color = int(np.median(edge_pixels))
    
    # If dark background (< 128), invert to white background for better OCR
    if bg_color < 128:
        gray = cv2.bitwise_not(gray)
        bg_color = 255  # After inversion, background is white

    # Add padding matching the background color
    padding = 30
    gray = cv2.copyMakeBorder(gray, padding, padding, padding, padding, 
                               cv2.BORDER_CONSTANT, value=bg_color)

    # Convert back to PIL format
    enhanced_image = Image.fromarray(gray)
    
    return enhanced_image


def upscale_and_set_dpi(image, scale_factor=2.0, method='bicubic', dpi=300):
    """Upscales the image using the specified method and sets the DPI"""
    
    # Convert PIL Image to OpenCV format
    img_arr = np.array(image)
    if len(img_arr.shape) == 2:
        # Grayscale
        img_cv = img_arr
    else:
        # RGB
        img_cv = cv2.cvtColor(img_arr, cv2.COLOR_RGB2BGR)
    
    new_width = int(img_cv.shape[1] * scale_factor)
    new_height = int(img_cv.shape[0] * scale_factor)
    dim = (new_width, new_height)

    if method == 'bicubic':
        upscaled_img_cv = cv2.resize(img_cv, dim, interpolation=cv2.INTER_CUBIC)
    elif method == 'lanczos':
        upscaled_img_cv = cv2.resize(img_cv, dim, interpolation=cv2.INTER_LANCZOS4)
    else:
        raise ValueError("Invalid upscaling method. Choose 'bicubic' or 'lanczos'.")

    # Convert back to PIL format
    if len(upscaled_img_cv.shape) == 2:
        upscaled_image = Image.fromarray(upscaled_img_cv)
    else:
        upscaled_image = Image.fromarray(cv2.cvtColor(upscaled_img_cv, cv2.COLOR_BGR2RGB))
    
    # Set DPI
    upscaled_image.info['dpi'] = (dpi, dpi)

    return upscaled_image

def ocr_me(image, image_path, raw_image=None):
    """
    Run OCR on the image. If the initial attempt fails (empty result),
    try fallback strategies including different PSM modes, thresholding, and scaling.
    """
    
    # helper to run ocr with specific config
    def run_tesseract_data(img_input, psm_mode):
        config = f"--psm {psm_mode} --oem 3"
        try:
            return pytesseract.image_to_data(img_input, lang='eng', config=config, output_type=pytesseract.Output.DICT)
        except Exception as e:
            print(f"OCR Error (PSM {psm_mode}): {e}")
            return {'text': [], 'line_num': [], 'block_num': [], 'left': []}

    def extract_text_from_data(data):
        # Group words by line number and reconstruct with indentation
        lines = {}
        # check if data has required keys
        if 'text' not in data or 'line_num' not in data:
            return ""
            
        for i, text in enumerate(data['text']):
            if text.strip():  # Skip empty entries
                line_num = data['line_num'][i]
                block_num = data['block_num'][i]
                key = (block_num, line_num)
                
                if key not in lines:
                    lines[key] = {'x': data['left'][i], 'words': []}
                lines[key]['words'].append(text)
        
        # Sort lines by block and line number
        sorted_keys = sorted(lines.keys())
        
        # Find minimum x position (leftmost text) to calculate relative indentation
        if lines:
            min_x = min(line['x'] for line in lines.values())
        else:
            return "" # No text found
        
        # Estimate character width based on image scale (2x upscale * typical char width)
        char_width = 16
        
        # Reconstruct text with indentation
        result_lines = []
        for key in sorted_keys:
            line = lines[key]
            # Calculate indentation based on x offset from minimum
            indent_pixels = line['x'] - min_x
            raw_indent = indent_pixels // char_width
            
            # Round to nearest 4 spaces (Python PEP 8 standard indentation)
            indent_spaces = round(raw_indent / 4) * 4
            
            # Join words with single space and add indentation
            line_text = ' '.join(line['words'])
            result_lines.append(' ' * indent_spaces + line_text)
        
        return '\n'.join(result_lines)

    # --- Strategy 1: Standard Attempt (Enhanced Image, PSM 6) ---
    data = run_tesseract_data(image, 6)
    ocr_text = extract_text_from_data(data)
    
    # --- Retry Loop if empty ---
    if not ocr_text.strip():
        print("Standard OCR failed. Starting fallback strategies...")
        
        # helper for simple string extraction
        def run_tesseract_string(img_input, psm_mode):
            config = f"--psm {psm_mode} --oem 3"
            try:
                return pytesseract.image_to_string(img_input, lang='eng', config=config)
            except Exception as e:
                print(f"OCR Error (String PSM {psm_mode}): {e}")
                return ""

        # List of strategies: (Description, Image_Preprocessor, PSM, Use_String_Method)
        strategies = [
            # 1. Try PSM 3 (Fully automatic) on original enhanced image
            ("PSM 3 Auto", lambda img: img, 3, False),
            
            # 2. Try PSM 4 (Single column) on original enhanced image
            ("PSM 4 Single Col", lambda img: img, 4, False),
            
            # 3. Binary Thresholding + PSM 6
            ("Binary Threshold + PSM 6", lambda img: apply_threshold(img), 6, False),
            
            # 4. Binary Thresholding + PSM 3
            ("Binary Threshold + PSM 3", lambda img: apply_threshold(img), 3, False),
            
            # 5. Stronger Upscale (3x) + PSM 6
            ("Upscale 3x + PSM 6", lambda img: upscale_and_set_dpi(raw_image if raw_image else Image.open(image_path), scale_factor=3.0, dpi=300), 6, False),

            # --- String Method Fallbacks (Less structured but may find text) ---
            
            # 6. String Method + PSM 3 (Auto)
            ("String Method + PSM 3", lambda img: img, 3, True),
            
            # 7. String Method + PSM 6 (Block)
            ("String Method + PSM 6", lambda img: img, 6, True),
            
            # 8. String Method + Binary Threshold + PSM 3
            ("String Method + Binary + PSM 3", lambda img: apply_threshold(img), 3, True),
            
             # 9. String Method + PSM 11 (Sparse text)
            ("String Method + PSM 11", lambda img: img, 11, True),
            
            # --- Desperation Fallbacks (Raw / Inverted) ---
            
            # 10. Raw Original Image (Bypass all enhancement)
            ("Raw Original + PSM 3", lambda img: raw_image if raw_image else Image.open(image_path), 3, True),
            
            # 11. Inverted Enhanced Image (In case BG detection failed)
            ("Inverted Enhanced + PSM 3", lambda img: ImageOps.invert(img.convert('L')), 3, True),
            
            # 12. Raw Original + Inverted
            ("Raw Inverted + PSM 3", lambda img: ImageOps.invert((raw_image if raw_image else Image.open(image_path)).convert('L')), 3, True),
        ]
        
        for name, process_func, psm, use_string in strategies:
            try:
                # Apply preprocessing
                processed_img = process_func(image)
                
                if use_string:
                    retry_text = run_tesseract_string(processed_img, psm)
                else:
                    retry_data = run_tesseract_data(processed_img, psm)
                    retry_text = extract_text_from_data(retry_data)
                
                if retry_text.strip():
                    print(f"Fallback '{name}' succeeded!")
                    ocr_text = retry_text
                    break
            except Exception as e:
                print(f"Fallback '{name}' failed: {e}")
                continue

    # Post-processing to fix common Python OCR errors
    ocr_text = fix_python_ocr_errors(ocr_text)
    
    # Clean up non-ASCII characters but preserve whitespace
    ocr_text = re.sub(r'[^\x00-\x7F]+', '', ocr_text)
    return ocr_text


def apply_threshold(image):
    """Apply binary thresholding to PIL image"""
    # Convert to CV2
    img_arr = np.array(image)
    if len(img_arr.shape) == 2:
        gray = img_arr
    else:
        gray = cv2.cvtColor(img_arr, cv2.COLOR_RGB2GRAY)
    
    # Apply Otsu's thresholding
    _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    
    # Convert back to PIL
    return Image.fromarray(binary)


def fix_python_ocr_errors(text):
    """Fix common OCR errors specific to Python code"""
    
    # GLOBAL FIX: Normalize corrupted quote characters
    # 'ô' is often misread as '"' or vice versa in some fonts/encodings
    text = text.replace('ô', '"')
    text = text.replace('”', '"') # Smart quotes
    text = text.replace('“', '"')
    text = text.replace('‘', "'")
    text = text.replace('’', "'")
    
    # Fix broken function names explicitly
    text = text.replace('def on_mouse down', 'def on_mouse_down')
    
    # GLOBAL FIX: 1 <-> l confusion
    
    # GLOBAL FIX: 1 <-> l confusion
    # When '1' appears between lowercase letters, it's almost always 'l'
    # e.g., "wind1ll" -> "windll", "se1f" -> "self"
    text = re.sub(r'([a-z])1([a-z])', r'\1l\2', text)
    
    # GLOBAL FIX: 0 <-> o confusion  
    # When '0' appears between lowercase letters, it's often 'o'
    # e.g., "c0de" -> "code"
    text = re.sub(r'([a-z])0([a-z])', r'\1o\2', text)
    
    # Cleanup: fix triple letters that result from over-correction
    # e.g., "windlll" -> "windll" (triple l is almost never valid)
    text = re.sub(r'lll+', 'll', text)
    text = re.sub(r'ooo+', 'oo', text)
    
    # Fix @ being misread as 0 in code contexts
    text = re.sub(r'\[@\]', '[0]', text)
    text = re.sub(r'\[@,', '[0,', text) # Slice [0, :]
    text = re.sub(r', @\]', ', 0]', text) # Slice [:, 0]
    text = re.sub(r'\(@\)', '(0)', text)
    text = re.sub(r'\(@,', '(0,', text)
    text = re.sub(r', @\)', ', 0)', text)
    text = re.sub(r', @,', ', 0,', text)
    text = re.sub(r'= @$', '= 0', text, flags=re.MULTILINE)
    text = re.sub(r'= @([,\)\s])', r'= 0\1', text)
    text = re.sub(r'(\d)@([,\)\s\]])', r'\1\2', text)

def process_image(image):
    """Process the captured image"""
    try:
        if image:
            raw_image = image.copy() # Keep a copy of original for fallbacks
            image = enhance_image(image)
            image_path = save_image(image)
            ocr_text = ocr_me(image, image_path, raw_image=raw_image)
            save_text(ocr_text)
            current_time = datetime.datetime.now().strftime("%H:%M:%S")
            print("[{}] OCR completed! Text copied to clipboard.".format(current_time))
        else:
            current_time = datetime.datetime.now().strftime("%H:%M:%S")
            print("[{}] No image to process.".format(current_time))
    except Exception as e:
        current_time = datetime.datetime.now().strftime("%H:%M:%S")
        print("[{}] Error processing image: {}".format(current_time, str(e)))


def start_snipping():
    """Start the snipping tool overlay"""
    # Run in a new thread to avoid blocking
    def run_snip():
        snip = SnippingTool(process_image)
        snip.start()
    
    thread = threading.Thread(target=run_snip)
    thread.daemon = True
    thread.start()


def on_pause_key():
    """Callback for when Pause key is pressed"""
    current_time = datetime.datetime.now().strftime("%H:%M:%S")
    print("[{}] Pause key pressed - starting snipping tool...".format(current_time))
    start_snipping()


def create_tray_icon():
    """Create the system tray icon image"""
    # Create a simple icon (64x64)
    icon_size = 64
    icon_image = PILImage.new('RGBA', (icon_size, icon_size), (0, 0, 0, 0))
    
    # Draw a simple circle with "T" for OCR/Text
    draw = ImageDraw.Draw(icon_image)
    # Blue background circle
    draw.ellipse([4, 4, icon_size-4, icon_size-4], fill=(30, 144, 255, 255))
    # Draw "T" for text/OCR
    draw.text((icon_size//2 - 8, icon_size//2 - 14), "T", fill=(255, 255, 255, 255))
    
    return icon_image


def reload_script(icon, item):
    """Reload the script"""
    global running
    running = False
    keyboard.unhook_all()
    icon.stop()
    os.execv(sys.executable, ['python'] + sys.argv)


def exit_app(icon, item):
    """Exit the application"""
    global running
    running = False
    keyboard.unhook_all()
    icon.stop()
    os._exit(0)


def setup_tray():
    """Setup and run the system tray icon"""
    global tray_icon
    
    menu = pystray.Menu(
        item('Reload', reload_script),
        item('Exit', exit_app)
    )
    
    tray_icon = pystray.Icon(
        "OCR Tool",
        create_tray_icon(),
        "OCR Tool (Press Ctrl+Scroll Lock to snip)",
        menu
    )
    
    return tray_icon


def main():
    """Main entry point"""
    global running
    
    print("OCR Tool started!")
    print("Press 'Ctrl+Scroll Lock' key to start snipping")
    print("Right-click system tray icon for options")
    
    # Register Ctrl+Scroll Lock as the hotkey with suppress to ensure it always triggers
    # suppress=True prevents the key from being passed to other applications
    keyboard.add_hotkey('ctrl+scroll lock', on_pause_key, suppress=True)
    
    # Setup and run system tray (this blocks)
    icon = setup_tray()
    icon.run()


if __name__ == "__main__":
    main()
